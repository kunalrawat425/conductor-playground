Vertex AI — Python Code

import vertexai
from vertexai.preview.vision_models import ImageGenerationModel

vertexai.init(project="YOUR_PROJECT_ID", location="us-central1")
model = ImageGenerationModel.from_pretrained("imagegeneration@006")

NEGATIVE = "ugly, distorted, blurry, price tags, text watermarks, cartoon, white background, fake smiles, stock photo poses, brand logos, plastic fish"

# Paste any prompt from this file
PROMPT = """[PASTE PROMPT HERE]"""

response = model.generate_images(
    prompt=PROMPT,
    negative_prompt=NEGATIVE,
    number_of_images=4,          # generate 4, pick best
    aspect_ratio="9:16",         # "9:16" Reel | "1:1" Feed | "16:9" FB
    safety_filter_level="block_few",
    person_generation="allow_adult",
    add_watermark=False,
    seed=42,                     # for reproducibility
)

for i, img in enumerate(response.images):
    img.save(f"asset_{i+1}.png")
    print(f"Saved asset_{i+1}.png")
